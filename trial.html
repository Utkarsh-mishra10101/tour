import os from flask import Flask, render_template_string, request, jsonify from google import genai from dotenv import load_dotenv load_dotenv() app = Flask(__name__) # Initialize the Gemini Client client = genai.Client(api_key=os.getenv("GEMINI_API_KEY")) # Create a multi-turn chat session using the Flash model chat = client.chats.create(model="gemini-2.5-flash") # HTML, CSS, and JS embedded in a single string HTML_TEMPLATE = """ Gemini Chatbot
Hello! How can I help you today?
""" @app.route("/") def index(): return render_template_string(HTML_TEMPLATE) @app.route("/chat", methods=["POST"]) def send_message(): user_message = request.json.get("message", "") if not user_message: return jsonify({"error": "Empty message"}), 400 try: response = chat.send_message(user_message) return jsonify({"reply": response.text}) except Exception as e: return jsonify({"error": str(e)}), 500 if __name__ == "__main__": app.run(debug=True, port=5000)
